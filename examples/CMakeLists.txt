function(xinn_example name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE xinn)
    target_compile_definitions(${name} PRIVATE XINN_DATA_DIR="${PROJECT_SOURCE_DIR}/data")
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endfunction()

xinn_example(bench_fusion)
xinn_example(spiral)
xinn_example(fundamental_diagram)
xinn_example(modules)
xinn_example(mnist_mlp)
xinn_example(traffic_state)
xinn_example(travel_time)
